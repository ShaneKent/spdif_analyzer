cmake_minimum_required(VERSION 3.21)
project(SpdifAnalyzer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SPDIF_BUILD_PLUGIN "Build Saleae plugin" ON)
option(SPDIF_BUILD_TESTS "Build portable tests" ON)
option(SPDIF_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(SPDIF_SDK_DIR "" CACHE PATH "Existing AnalyzerSDK checkout (optional)")
add_library(spdif_core STATIC src/core/Decoder.cpp)
set_target_properties(spdif_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(spdif_core PUBLIC src)
if(MSVC)
 target_compile_options(spdif_core PRIVATE /W4)
else()
 target_compile_options(spdif_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SPDIF_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
 target_compile_options(spdif_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(SPDIF_BUILD_PLUGIN)
 if(NOT SPDIF_SDK_DIR)
  include(FetchContent)
  FetchContent_Declare(analyzersdk GIT_REPOSITORY https://github.com/saleae/AnalyzerSDK.git GIT_TAG e3c4b4ffda1c162c024a5913520aed17c6edb238)
  FetchContent_GetProperties(analyzersdk)
  if(NOT analyzersdk_POPULATED)
   FetchContent_Populate(analyzersdk)
  endif()
  set(SPDIF_SDK_DIR "${analyzersdk_SOURCE_DIR}")
 endif()
 include("${SPDIF_SDK_DIR}/AnalyzerSDKConfig.cmake")
 file(GLOB ADAPTER_SOURCES CONFIGURE_DEPENDS src/saleae/*.cpp)
 add_library(SpdifAnalyzer MODULE ${ADAPTER_SOURCES})
 target_compile_definitions(SpdifAnalyzer PRIVATE LOGIC2)
 target_link_libraries(SpdifAnalyzer PRIVATE spdif_core Saleae::AnalyzerSDK)
 set_target_properties(SpdifAnalyzer PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Analyzers" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Analyzers")
 install(TARGETS SpdifAnalyzer LIBRARY DESTINATION Analyzers RUNTIME DESTINATION Analyzers)
endif()
if(SPDIF_BUILD_TESTS)
 enable_testing()
 add_executable(test_decoder tests/test_decoder.cpp)
 target_link_libraries(test_decoder PRIVATE spdif_core)
 add_test(NAME decoder COMMAND test_decoder)
endif()
