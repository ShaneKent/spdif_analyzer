cmake_minimum_required(VERSION 3.21)
project(SpdifAnalyzer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SPDIF_BUILD_PLUGIN "Build Saleae plugin" ON)
option(SPDIF_BUILD_TESTS "Build portable tests" ON)
option(SPDIF_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(SPDIF_SDK_DIR "" CACHE PATH "Existing AnalyzerSDK checkout (optional)")
add_library(spdif_core STATIC src/core/Decoder.cpp src/core/Status.cpp)
set_target_properties(spdif_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(spdif_core PUBLIC src src/core)
if(MSVC)
 target_compile_options(spdif_core PRIVATE /W4)
else()
 target_compile_options(spdif_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(spdif_analysis STATIC src/analysis/Analysis.cpp)
target_link_libraries(spdif_analysis PUBLIC spdif_core)
set_target_properties(spdif_analysis PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(spdif_exports STATIC src/export/Exports.cpp)
target_link_libraries(spdif_exports PUBLIC spdif_core)
set_target_properties(spdif_exports PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(SPDIF_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
 foreach(target spdif_core spdif_analysis spdif_exports)
  target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 endforeach()
endif()
if(SPDIF_BUILD_PLUGIN)
 if(NOT SPDIF_SDK_DIR)
  include(FetchContent)
  FetchContent_Declare(analyzersdk GIT_REPOSITORY https://github.com/saleae/AnalyzerSDK.git GIT_TAG e3c4b4ffda1c162c024a5913520aed17c6edb238)
  FetchContent_GetProperties(analyzersdk)
  if(NOT analyzersdk_POPULATED)
   FetchContent_Populate(analyzersdk)
  endif()
  set(SPDIF_SDK_DIR "${analyzersdk_SOURCE_DIR}")
 endif()
 include("${SPDIF_SDK_DIR}/AnalyzerSDKConfig.cmake")
 file(GLOB ADAPTER_SOURCES CONFIGURE_DEPENDS src/saleae/*.cpp)
 add_library(SpdifAnalyzer MODULE ${ADAPTER_SOURCES})
 target_compile_definitions(SpdifAnalyzer PRIVATE LOGIC2)
 if(MSVC)
  target_compile_options(SpdifAnalyzer PRIVATE /W4 /utf-8)
 endif()
 target_link_libraries(SpdifAnalyzer PRIVATE spdif_core spdif_analysis spdif_exports Saleae::AnalyzerSDK)
 set_target_properties(SpdifAnalyzer PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Analyzers" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Analyzers")
 if(WIN32)
  add_executable(smoke_windows scripts/smoke_windows.cpp)
  target_compile_definitions(smoke_windows PRIVATE LOGIC2)
  target_link_libraries(smoke_windows PRIVATE Saleae::AnalyzerSDK)
 endif()
 if(SPDIF_BUILD_TESTS)
  add_library(SpdifTestSignal MODULE tests/SignalSource.cpp)
  target_compile_definitions(SpdifTestSignal PRIVATE LOGIC2)
  target_link_libraries(SpdifTestSignal PRIVATE Saleae::AnalyzerSDK)
  set_target_properties(SpdifTestSignal PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/TestAnalyzers" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/TestAnalyzers")
 endif()
 install(TARGETS SpdifAnalyzer LIBRARY DESTINATION Analyzers RUNTIME DESTINATION Analyzers)
endif()
if(SPDIF_BUILD_TESTS)
 enable_testing()
 add_executable(test_decoder tests/test_decoder.cpp)
 target_link_libraries(test_decoder PRIVATE spdif_core)
 add_test(NAME decoder COMMAND test_decoder)
 add_executable(benchmark_decoder tests/benchmark.cpp)
 target_link_libraries(benchmark_decoder PRIVATE spdif_core)
endif()

add_executable(decode_edges tools/decode_edges.cpp)
target_link_libraries(decode_edges PRIVATE spdif_core spdif_analysis spdif_exports)

if(SPDIF_BUILD_TESTS)
 foreach(name protocol_features analysis exports)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_include_directories(test_${name} PRIVATE tests)
  target_link_libraries(test_${name} PRIVATE spdif_core spdif_analysis spdif_exports)
  add_test(NAME ${name} COMMAND test_${name})
 endforeach()
 find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:decode_edges>)
  add_test(NAME compare COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_compare.py)
 endif()
endif()
